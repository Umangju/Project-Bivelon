const ADMIN_KEY = "0326"; // 설정하신 비밀번호

function doGet(e) {
  var key = e.parameter.key;
  var callback = e.parameter.callback; // 로컬 파일 보안 우회용
  var responseObj;

  if (key !== ADMIN_KEY) {
    responseObj = { result: "error", message: "비밀번호가 일치하지 않습니다." };
  } else {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var rows = sheet.getDataRange().getValues();
    var data = [];

    // 시트에 2행 이상 데이터가 있을 때만 읽기 (1행만 있거나 비어있어도 정상 통과)
    if (rows.length > 1) {
      for (var i = 1; i < rows.length; i++) {
        var r = rows[i];
        // 완전 빈 줄은 건너뛰기
        if (!r[0] && !r[1] && !r[2] && !r[3]) continue;

        var dateVal = r[0];
        if (dateVal instanceof Date) {
          dateVal = Utilities.formatDate(dateVal, "GMT+9", "yyyy-MM-dd HH:mm:ss");
        }
        data.push({
          rowNumber: i + 1,
          date: String(dateVal || ""),
          company: String(r[1] || ""),
          owner: String(r[2] || ""),
          phone: String(r[3] || ""),
          region: String(r[4] || ""),
          biz_type: String(r[5] || ""),
          sales: String(r[6] || ""),
          fund: String(r[7] || ""),
          status: String(r[8] || ""),
          message: String(r[9] || ""),
          diag_industry: String(r[10] || ""),
          diag_age: String(r[11] || ""),
          diag_sales: String(r[12] || ""),
          diag_purpose: String(r[13] || ""),
          diag_amount: String(r[14] || ""),
          diag_region: String(r[15] || ""),
          diag_main: String(r[16] || ""),
          diag_tip: String(r[17] || ""),
          diag_score: String(r[18] || ""),
          diag_secondary: String(r[19] || ""),
          diag_summary: String(r[20] || ""),
          admin_status: String(r[21] || "상담대기"),
          admin_memo: String(r[22] || "")
        });
      }
      data.reverse();
    }
    responseObj = { result: "success", data: data };
  }

  var jsonString = JSON.stringify(responseObj);

  // 로컬 파일(admin.html)에서 요청했을 때 보안 차단을 통과하도록 처리
  if (callback) {
    return ContentService.createTextOutput(callback + "(" + jsonString + ");")
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }

  return ContentService.createTextOutput(jsonString)
    .setMimeType(ContentService.MimeType.JSON);
}

// 2. 고객이 상담 신청을 하거나, 관리자가 상태/메모를 수정할 때 실행되는 함수
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = e.parameter;

  // [관리자 기능] 상담 상태 및 메모 업데이트 요청인 경우
  if (data.action === "update") {
    if (data.key !== ADMIN_KEY) {
      return ContentService.createTextOutput(JSON.stringify({ result: "error", message: "권한이 없습니다." }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    var rowNum = parseInt(data.rowNumber, 10);
    if (rowNum > 1) {
      sheet.getRange(rowNum, 22).setValue(data.admin_status || "상담대기"); // V열(22번째): 처리상태
      sheet.getRange(rowNum, 23).setValue(data.admin_memo || "");          // W열(23번째): 관리자메모
    }
    return ContentService.createTextOutput(JSON.stringify({ result: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // [관리자 기능] 행 삭제 요청인 경우
  if (data.action === "delete") {
    if (data.key !== ADMIN_KEY) {
      return ContentService.createTextOutput(JSON.stringify({ result: "error", message: "권한이 없습니다." }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    var delRowNum = parseInt(data.rowNumber, 10);
    if (delRowNum > 1) {
      sheet.deleteRow(delRowNum);
    }
    return ContentService.createTextOutput(JSON.stringify({ result: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // [고객 기능] 홈페이지에서 신규 상담 신청이 들어온 경우
  var now = new Date();
  var date = Utilities.formatDate(now, "GMT+9", "yyyy-MM-dd HH:mm:ss");

  sheet.appendRow([
    date,
    data.company || "",
    data.owner || "",
    "'" + (data.phone || ""), // 앞자리 0이 사라지지 않도록 작은따옴표 처리
    data.region || "",
    data.biz_type || "",
    data.sales || "",
    data.fund || "",
    data.status || "",
    data.message || "",
    data.diagnosis_industry || "",
    data.diagnosis_age || "",
    data.diagnosis_sales || "",
    data.diagnosis_purpose || "",
    data.diagnosis_amount || "",
    data.diagnosis_region || "",
    data.diagnosis_main || "",
    data.diagnosis_tip || "",
    data.diagnosis_score || "",
    data.diagnosis_secondary || "",
    data.diagnosis_summary || "",
    "상담대기", // V열 기본값
    ""          // W열 기본값
  ]);

  return ContentService.createTextOutput(JSON.stringify({ result: "success" }))
    .setMimeType(ContentService.MimeType.JSON);
}
